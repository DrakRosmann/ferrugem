# Controle de jobs: segundo plano, Ctrl+Z, Ctrl+C, fg, bg, kill %n.

$ sleep 30 &
[1] *

$ jobs
[1]  Executando  sleep 30

$ kill %1
$ jobs
[1]  Terminado   sleep 30

$ yes > /dev/null
<Ctrl+Z>
^Z
[1]  Parado      yes >/dev/null

$ jobs
[1]  Parado      yes >/dev/null

$ bg
[1] yes >/dev/null &

$ kill -9 %1
$ jobs
[1]  Morto       yes >/dev/null

$ sleep 30
<Ctrl+C>
^C

$ echo $?
130

$ sleep 1 && echo acordou &
[1] *

$ wait; jobs
acordou
[1]  Concluído   sleep 1 && echo acordou

$ sleep 30
<Ctrl+Z>
^Z
[1]  Parado      sleep 30

$ fg
<Ctrl+C>
sleep 30
^C

$ echo $?
130
