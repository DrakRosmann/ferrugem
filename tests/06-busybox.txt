# O busybox (compilado com a musl por scripts/busybox.sh): os comandos Unix
# de verdade rodando sobre as syscalls do FerrugemOS.

$ busybox | head -n 1
BusyBox v1.37.0 (*) multi-call binary.

$ seq 1 10 | awk '{s+=$1} END {print "soma", s}'
soma 55

$ echo 'o rato roeu' | sed 's/rato/gato/; s/roeu/dormiu/'
o gato dormiu

# Acentos (UTF-8) passam pelo terminal e pelos pipes.
$ printf 'banana\nmaçã\nbanana\nuva\n' | sort | uniq -c
      2 banana
      1 maçã
      1 uva

$ find /etc -type f | sort
/etc/hostname
/etc/motd

$ mkdir -p /tmp/x/y; echo dado > /tmp/x/y/z; cp -r /tmp/x /tmp/w; cat /tmp/w/y/z; rm -rf /tmp/x /tmp/w; ls /tmp
dado

$ tar -czf /tmp/etc.tgz -C / etc && tar -tzf /tmp/etc.tgz | sort
etc/
etc/hostname
etc/motd

$ echo abc | md5sum
0bee89b07a248e27c83fc3d5951213c1  -

$ dd if=/dev/zero bs=1K count=64 2>/dev/null | wc -c
65536

# O ash: subshells, $(...), aritmética, funções, case, trap, wait (que usa
# rt_sigsuspend).
$ ash -c 'x=$(echo sub); echo "[$x] $((6*7))"; for f in a b; do echo item $f; done'
[sub] 42
item a
item b

$ ash -c 'f() { echo "função $1"; }; f ok; case xyz in x*) echo casou;; esac'
função ok
casou

$ ash -c 'trap "echo pegou USR1" USR1; kill -USR1 $$; echo continuou'
pegou USR1
continuou

$ ash -c 'sleep 1 & wait $!; echo esperou $?'
esperou 0

# chmod, umask e chown.
$ touch f; chmod 751 f; stat -c '%a' f; umask 077; touch g; stat -c '%a' g; umask; umask 022
751
600
0077

$ chown 1000:100 f; stat -c '%u %g' f; rm f g
1000 100

# poll com prazo: o timeout encerra o cat que espera o teclado.
$ timeout 1 cat; echo $?
Terminado
143

$ uptime | grep -c 'load average'
1

> \e
> :wq\r

segunda linha

...
